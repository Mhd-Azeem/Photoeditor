cmake_minimum_required(VERSION 3.22.1)
project(az_upscaler)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED NCNN_ROOT)
  set(NCNN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/ncnn")
endif()
set(ncnn_DIR "${NCNN_ROOT}/${ANDROID_ABI}/lib/cmake/ncnn")
find_package(ncnn REQUIRED)
# NCNN's prebuilt package exports -fno-exceptions; JNI needs allocation error handling.
get_target_property(NCNN_OPTIONS ncnn INTERFACE_COMPILE_OPTIONS)
if(NCNN_OPTIONS)
  list(REMOVE_ITEM NCNN_OPTIONS "-fno-exceptions")
  set_target_properties(ncnn PROPERTIES INTERFACE_COMPILE_OPTIONS "${NCNN_OPTIONS}")
endif()
add_library(az_upscaler SHARED upscaler.cpp)
target_compile_options(az_upscaler PRIVATE -fexceptions)
target_link_libraries(az_upscaler ncnn jnigraphics android log)
target_link_options(az_upscaler PRIVATE "-Wl,-z,max-page-size=16384")
